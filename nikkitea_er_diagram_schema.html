Relational Schema
Table Primary Key Foreign Key(s) Description
users user_id - Login and role records for cashier/admin.
categories category_id - Menu categories such as Milk Tea, Fruit Tea, Snacks, Add-ons.
products product_id category_id → categories.category_id Menu item details and base price.
inventory product_id product_id → products.product_id Current stock and low-stock threshold.
orders order_id user_id → users.user_id Order transaction and order type.
order_items order_item_id order_id → orders.order_id, product_id → products.product_id Each item on an order, including quantity and customization.
add_ons addon_id - Reusable add-on items that can be added to a product.
order_item_addons order_item_addon_id order_item_id → order_items.order_item_id, addon_id → add_ons.addon_id Selected add-ons per line item.
payments payment_id order_id → orders.order_id Cash amount and change amount collected.
sales sale_id order_id → orders.order_id Receipt and completed sales history.
CREATE TABLE users ( user_id SERIAL PRIMARY KEY, username VARCHAR(50) UNIQUE NOT NULL, password_hash VARCHAR(255) NOT NULL, full_name VARCHAR(100) NOT NULL, role VARCHAR(20) NOT NULL CHECK (role IN ('admin', 'cashier')) ); CREATE TABLE categories ( category_id SERIAL PRIMARY KEY, category_name VARCHAR(50) UNIQUE NOT NULL ); CREATE TABLE products ( product_id SERIAL PRIMARY KEY, category_id INT NOT NULL, product_code VARCHAR(30) UNIQUE NOT NULL, product_name VARCHAR(100) NOT NULL, base_price NUMERIC(10,2) NOT NULL, FOREIGN KEY (category_id) REFERENCES categories(category_id) ); CREATE TABLE inventory ( product_id INT PRIMARY KEY, current_stock INT NOT NULL DEFAULT 0, low_stock_threshold INT NOT NULL DEFAULT 5, FOREIGN KEY (product_id) REFERENCES products(product_id) ); CREATE TABLE orders ( order_id SERIAL PRIMARY KEY, user_id INT NOT NULL, order_type VARCHAR(20) NOT NULL CHECK (order_type IN ('DINE_IN', 'TAKE_OUT')), status VARCHAR(20) DEFAULT 'COMPLETED', total_amount NUMERIC(10,2) NOT NULL DEFAULT 0, FOREIGN KEY (user_id) REFERENCES users(user_id) ); CREATE TABLE order_items ( order_item_id SERIAL PRIMARY KEY, order_id INT NOT NULL, product_id INT NOT NULL, quantity INT NOT NULL CHECK (quantity > 0), sugar_level VARCHAR(20) DEFAULT 'N/A', cup_size VARCHAR(20) DEFAULT 'N/A', unit_price NUMERIC(10,2) NOT NULL, FOREIGN KEY (order_id) REFERENCES orders(order_id), FOREIGN KEY (product_id) REFERENCES products(product_id) ); CREATE TABLE add_ons ( addon_id SERIAL PRIMARY KEY, addon_name VARCHAR(80) UNIQUE NOT NULL, addon_price NUMERIC(10,2) NOT NULL DEFAULT 0 ); CREATE TABLE order_item_addons ( order_item_addon_id SERIAL PRIMARY KEY, order_item_id INT NOT NULL, addon_id INT NOT NULL, addon_price NUMERIC(10,2) NOT NULL, FOREIGN KEY (order_item_id) REFERENCES order_items(order_item_id), FOREIGN KEY (addon_id) REFERENCES add_ons(addon_id) ); CREATE TABLE payments ( payment_id SERIAL PRIMARY KEY, order_id INT NOT NULL, amount_paid NUMERIC(10,2) NOT NULL, change_amount NUMERIC(10,2) NOT NULL DEFAULT 0, FOREIGN KEY (order_id) REFERENCES orders(order_id) ); CREATE TABLE sales ( sale_id SERIAL PRIMARY KEY, order_id INT NOT NULL, receipt TEXT, total NUMERIC(10,2) NOT NULL, completed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (order_id) REFERENCES orders(order_id) );
