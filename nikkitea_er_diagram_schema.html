Relational Schema

Table Primary Key Foreign Key(s) Notes
users user_id - Staff/admin login and role info.
categories category_id - Groups menu items such as Milk Tea, Fruit Tea, Snacks, Add-ons.
products product_id category_id → categories.category_id Stores product code, name, and base price.
inventory product_id product_id → products.product_id Tracks current stock and low-stock threshold.
orders order_id user_id → users.user_id Captures order type, status, totals, and payment data.
order_items order_item_id order_id → orders.order_id, product_id → products.product_id Stores quantity, customization, and item price at purchase time.
add_ons addon_id - Reusable add-on menu such as Pearl, Nata, Cheese, etc.
order_item_addons order_item_addon_id order_item_id → order_items.order_item_id, addon_id → add_ons.addon_id Tracks selected add-ons per order line item.
payments payment_id order_id → orders.order_id Stores cash amount and change given.
sales sale_id order_id → orders.order_id Stores receipt and completed sales history for the app dashboard.
CREATE TABLE users ( user_id SERIAL PRIMARY KEY, username VARCHAR(50) UNIQUE NOT NULL, password_hash VARCHAR(255) NOT NULL, full_name VARCHAR(100) NOT NULL, role VARCHAR(20) NOT NULL CHECK (role IN ('admin', 'cashier')) ); CREATE TABLE categories ( category_id SERIAL PRIMARY KEY, category_name VARCHAR(50) UNIQUE NOT NULL ); CREATE TABLE products ( product_id SERIAL PRIMARY KEY, category_id INT NOT NULL, product_code VARCHAR(30) UNIQUE NOT NULL, product_name VARCHAR(100) NOT NULL, base_price NUMERIC(10,2) NOT NULL, image_file VARCHAR(200), is_active BOOLEAN DEFAULT TRUE, FOREIGN KEY (category_id) REFERENCES categories(category_id) ); CREATE TABLE inventory ( product_id INT PRIMARY KEY, current_stock INT NOT NULL DEFAULT 0, low_stock_threshold INT NOT NULL DEFAULT 5, last_updated TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (product_id) REFERENCES products(product_id) ); CREATE TABLE orders ( order_id SERIAL PRIMARY KEY, user_id INT NOT NULL, order_type VARCHAR(20) NOT NULL CHECK (order_type IN ('DINE_IN', 'TAKE_OUT')), status VARCHAR(20) DEFAULT 'COMPLETED', order_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP, total_amount NUMERIC(10,2) NOT NULL DEFAULT 0, FOREIGN KEY (user_id) REFERENCES users(user_id) ); CREATE TABLE order_items ( order_item_id SERIAL PRIMARY KEY, order_id INT NOT NULL, product_id INT NOT NULL, quantity INT NOT NULL CHECK (quantity > 0), sugar_level VARCHAR(20) DEFAULT 'N/A', cup_size VARCHAR(20) DEFAULT 'N/A', unit_price NUMERIC(10,2) NOT NULL, FOREIGN KEY (order_id) REFERENCES orders(order_id), FOREIGN KEY (product_id) REFERENCES products(product_id) ); CREATE TABLE add_ons ( addon_id SERIAL PRIMARY KEY, addon_name VARCHAR(80) UNIQUE NOT NULL, addon_price NUMERIC(10,2) NOT NULL DEFAULT 0 ); CREATE TABLE order_item_addons ( order_item_addon_id SERIAL PRIMARY KEY, order_item_id INT NOT NULL, addon_id INT NOT NULL, addon_price NUMERIC(10,2) NOT NULL, FOREIGN KEY (order_item_id) REFERENCES order_items(order_item_id), FOREIGN KEY (addon_id) REFERENCES add_ons(addon_id) ); CREATE TABLE payments ( payment_id SERIAL PRIMARY KEY, order_id INT NOT NULL, amount_paid NUMERIC(10,2) NOT NULL, change_amount NUMERIC(10,2) NOT NULL DEFAULT 0, paid_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (order_id) REFERENCES orders(order_id) ); CREATE TABLE sales ( sale_id SERIAL PRIMARY KEY, order_id INT NOT NULL, receipt TEXT, total NUMERIC(10,2) NOT NULL, order_type VARCHAR(20) NOT NULL, completed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (order_id) REFERENCES orders(order_id) );
